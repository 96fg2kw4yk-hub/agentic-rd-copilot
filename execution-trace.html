# repo_indexer.py 第 107-151 行
def scan_and_parse(self, repo_id, local_path):
    self.current_repo_id = "e26671e73d18"
    self.chunks = []         # 清空旧数据

    # ① 找文件
    py_files = list(Path(local_path).rglob("*.py"))
    # → [requests/__init__.py, requests/api.py, requests/auth.py, ...]
    #   可能几百个文件

    # ② 过滤
    exclude_dirs = {"venv", ".git", "__pycache__", ...}
    # 排除虚拟环境和缓存目录

    valid_files = [f for f in py_files if f.stat().st_size/1024 <= 500]
    # 排除 >500KB 的大文件

    # ③ 逐个 AST 解析
    for file_path in valid_files:
        source = file_path.read_text(encoding="utf-8", errors="ignore")
        file_chunks = self.ast_parser.parse_file(repo_id, str(file_path), source)
        self.chunks.extend(file_chunks)
        # 每个 .py 文件产生 N 个 chunk（1 文件级 + 若干函数/类）

为什么每个 .py 文件至少产生一个 chunk？ 即使文件里没有函数或类（比如只有 import 语句的 __init__.py），parse_file 也会生成一个文件级 chunk（symbol_type="file"）。这保证了检索时不会漏掉任何文件。

Step 2b: build_vector_index() — ChromaDB 向量化

# repo_indexer.py 第 155-197 行
def build_vector_index(self):
    # ① CodeChunk → LangChain Document
    docs = []
    for chunk in self.chunks:
        docs.append(Document(
            page_content = self._chunk_to_text(chunk),
            # page_content 是一段拼接文本：
            #   "[function] validate_token\n"
            #   "文件: requests/auth.py\n"
            #   "摘要: 函数 validate_token(self, token)...\n"
            #   "文档: 验证 JWT token 的有效性\n"
            #   "依赖: jwt, time, hashlib\n"
            #   "参数: self, token\n"
            #   "代码:\n"
            #   "def validate_token(self, token):\n ..."

            metadata = {
                "chunk_id": chunk.chunk_id,        # 全局唯一
                "file_path": chunk.file_path,       # requests/auth.py
                "symbol_type": chunk.symbol_type,   # function
                "symbol_name": chunk.symbol_name,   # validate_token
                "start_line": chunk.start_line,     # 42
                "end_line": chunk.end_line,         # 58
            }
        ))

    # ② 调用 Ollama embedding，为每个 doc 生成向量
    print(f"  [index] 正在嵌入 {len(docs)} 个文档...")
    # 这一步最慢！每个 doc 调一次 Ollama API → nomic-embed-text 模型 → 768 维向量
    # 几千个 doc 可能需要几分钟

    self.vector_store = Chroma.from_documents(
        documents=docs,
        embedding=self.embeddings,                    # OllamaEmbeddings(model="nomic-embed-text")
        persist_directory=".../data/indexes/chroma/repo_e26671e73d18",
    )
    # ChromaDB 内部用 HNSW 算法建索引，查询时做近似最近邻搜索
性能瓶颈在这里！每个 chunk 调用一次 Ollama API 生成 embedding，即使是本地运行，几千个 chunk 也要几分钟。后续优化方向：批量 embedding（一次调 API 送多个 chunk）、增量索引（只对新文件做 embedding）。

Step 2c: build_bm25_index() — BM25 倒排索引

# repo_indexer.py 第 201-221 行
def build_bm25_index(self):
    tokenized = []
    self.bm25_chunks = []

    for chunk in self.chunks:
        text = self._chunk_to_text(chunk)       # 同上的拼接文本
        tokens = list(jieba.cut(text))          # jieba 中文分词
        # "JWT decode failed" →
        #   ["JWT", "decode", "failed"]          (英文原样保留)
        # "验证token有效性" →
        #   ["验证", "token", "有效性"]           (中文被切分)

        tokenized.append(tokens)
        self.bm25_chunks.append(chunk)

    self.bm25 = BM25Okapi(tokenized)
    # BM25Okapi 内部：
    #   1. 统计每个词的文档频率（IDF）
    #   2. 统计每个词在每篇文档中的词频（TF）
    #   3. 查询时用 BM25 公式打分

    # 持久化到磁盘
    pickle.dump({"chunks": self.bm25_chunks, "bm25": self.bm25},
                open(".../data/indexes/bm25_e26671e73d18.pkl", "wb"))
BM25 和向量索引的关键区别：BM25 不依赖任何外部服务，纯 Python 计算，几秒钟就完成。而向量索引需要 Ollama 逐个生成 embedding，慢得多。这也是为什么需要混合检索——如果 LLM 挂了，BM25 至少还能工作（fallback 思维）。

几个容易被忽略的执行细节

① hybrid_search 每次都要从磁盘加载索引
_load_vector_store() 和 _load_bm25() 每次查询都从磁盘重新加载。这不是最优的——如果服务常驻，应该把索引缓存在内存里。但 V1 这样设计是正确的：先跑通，再优化性能。过早优化是万恶之源。
② LangGraph 的 invoke 是同步阻塞的
6 个节点依次执行（其中 4 个要调 LLM），整个流程可能需要 20-60 秒。在这期间 FastAPI 的请求处理线程被阻塞。虽然 task 是异步创建的（先返回 task_id），但 run_full_pipeline 本身在主线程中同步执行，这意味着同时只能处理一个 Issue。换成 BackgroundTasks 或 asyncio.to_thread 可以解决。
③ 每个 Agent 的 LLM 调用都是独立的
IssueAnalyzer、BugLocator、PatchGenerator、PRAgent 各自实例化自己的 ChatOllama 对象，互不共享上下文。这保证了每个 Agent 的 prompt 是干净、独立的，不会被前一个 Agent 的输出"污染"。但也意味着不能用 conversation history（每轮都从零开始），对于需要多轮推理的场景，这是需要改进的。
④ 索引只用一次的数据结构在内存中不清理
self.chunks 和 self.bm25_chunks 在 scan_and_parse 后一直驻留内存，直到服务重启。对于小仓库没问题；对于大仓库，应该考虑索引持久化后释放 chunks 原始数据，只保留 vector_store 和 bm25 两个查询接口。
